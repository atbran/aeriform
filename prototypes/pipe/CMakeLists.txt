cmake_minimum_required(VERSION 3.22)
project(PipePrototype LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(pipe-verify verify.cpp)
add_executable(pipe-render render.cpp)
if(MINGW)
    target_link_options(pipe-verify PRIVATE -static)
    target_link_options(pipe-render PRIVATE -static)
endif()
foreach(target pipe-verify pipe-render)
    if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endforeach()
